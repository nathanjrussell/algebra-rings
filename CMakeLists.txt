cmake_minimum_required(VERSION 3.24)
project(njr_algebra_rings VERSION 0.1.0 LANGUAGES CXX)
option(NJR_ALGEBRA_RINGS_BUILD_EXAMPLES "Build examples" ${PROJECT_IS_TOP_LEVEL})
option(NJR_ALGEBRA_RINGS_BUILD_TESTS "Build tests" ${PROJECT_IS_TOP_LEVEL})
include(cmake/Dependencies.cmake)
add_subdirectory(src)
if(NJR_ALGEBRA_RINGS_BUILD_EXAMPLES OR NJR_ALGEBRA_RINGS_BUILD_TESTS)
    include(cmake/DevelopmentDependencies.cmake)
endif()
if(NJR_ALGEBRA_RINGS_BUILD_EXAMPLES)
    add_subdirectory(example)
endif()
if(NJR_ALGEBRA_RINGS_BUILD_TESTS)
    enable_testing()
    include(cmake/GoogleTest.cmake)
    add_subdirectory(test)
endif()
